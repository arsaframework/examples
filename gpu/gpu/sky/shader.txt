{
	setportrait()
	screensize(640,1351, true)
	set( $sid, gpu.addshader(ftostring("vs.txt"), ftostring("sky.txt")) )


	set($name, "Layer 1" )
	setmaterialtype($name, $sid)
	settexturelayer($name, iChannel0, 0)
	settexturelayer($name, iChannel1, 1)
	settextureclamp($name, repeat)

	set($t,0.0)
}function(init)

{
	gpu.removeshader($sid)
}function(deinit)

gpu.setconst($name,"iChannel0", 0)
gpu.setconst($name,"iChannel1", 1)

add($t,getdeltatime())
gpu.setconst($name,"iTime", $t)

set($sw, getscreensizewidth())
set($sh, getscreensizeheight())
gpu.setconst($name,"iResolution", $sw, $sh, 1)

set($tx, touch("gettouchx") )
set($ty, touch("gettouchy") )
gpu.setconst($name,"iMouse", 320, 480, 1.0, 0.0)
